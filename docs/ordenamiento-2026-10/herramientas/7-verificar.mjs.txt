// SOLO LECTURA. Verifica la copia ordenada contra el plan: cada archivo en su destino con su peso, ninguno en su
// lugar viejo, el total de archivos igual que antes, y la forma de cada cliente (carpetas de primer nivel del manual).
import fs from 'node:fs';
import path from 'node:path';

const BASE = 'D:/ORDENADO/CLIENTES EFFORT E.A.S/CLIENTES';
const plan = JSON.parse(fs.readFileSync('plan.json', 'utf8'));
let bien = 0; const mal = [];
for (const f of plan) {
  const a = path.join(BASE, f.destinoRel);
  let st = null; try { st = fs.statSync(a); } catch {}
  if (!st) { mal.push(['no está en el destino', f.destinoRel]); continue; }
  if (st.size !== f.tam) { mal.push([`pesa ${st.size} y debía ${f.tam}`, f.destinoRel]); continue; }
  if (f.cambia && f.origen.toLowerCase() !== f.destinoRel.toLowerCase() && fs.existsSync(path.join(BASE, f.origen))) { mal.push(['sigue también en el lugar viejo', f.origen]); continue; }
  bien++;
}
let total = 0;
const contar = (d) => { for (const e of fs.readdirSync(d, { withFileTypes: true })) e.isDirectory() ? contar(path.join(d, e.name)) : total++; };
contar('D:/ORDENADO');
// Forma de cada cliente
const raros = [];
for (const c of fs.readdirSync(BASE, { withFileTypes: true }).filter((d) => d.isDirectory() && /^\d{3}(?!\d)/.test(d.name) && !d.name.startsWith('999'))) {
  const cod = c.name.slice(0, 3);
  for (const e of fs.readdirSync(path.join(BASE, c.name), { withFileTypes: true })) {
    const ok = e.isDirectory() && (new RegExp(`^${cod} (LEGAL Y SOCIETARIO|TRIBUTARIO|ZZ A CLASIFICAR|PERIODO \\d{4})$`).test(e.name));
    if (!ok) raros.push(`${c.name}/${e.name}${e.isDirectory() ? '/' : ''}`);
  }
}
console.log(`plan: ${plan.length} · bien: ${bien} · mal: ${mal.length}`);
console.log(`archivos en D:\\ORDENADO: ${total} (antes 36488)`);
console.log(`elementos de primer nivel fuera del manual en los clientes: ${raros.length}`, raros.slice(0, 20));
if (mal.length) console.log(mal.slice(0, 20));
fs.writeFileSync('verificacion-ordenado.json', JSON.stringify({ fecha: new Date().toISOString(), plan: plan.length, bien, mal, totalArchivos: total, raros }, null, 1));
