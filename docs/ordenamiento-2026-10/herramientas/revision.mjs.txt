// BUCLE DE REVISIÓN del ordenamiento en el OneDrive real: verifica contra el PLAN, contra la COPIA DE SEGURIDAD y
// contra la ESTRUCTURA DEL 999, corrige lo que encuentre y vuelve a revisar. Mínimo 3 rondas; termina cuando una
// ronda (de la 3 en adelante) sale con 0 problemas y 0 correcciones. Máximo 8 rondas.

// Correcciones posibles (mismas protecciones que el aplicador: sin DELETE, conflictBehavior=fail, If-Match, diario):
//   a) un archivo del plan que no está en su destino (y su contenido es el del plan) → se mueve al destino del plan;
//   b) un archivo fuera de la estructura del 999 dentro de un cliente → a «NNN ZZ A CLASIFICAR/<ruta>» sin renombrar;
//   c) un duplicado que sigue dentro de un cliente → a «ZZ DUPLICADOS PARA REVISAR/<ruta original>»;
//   d) una carpeta vacía dentro de un cliente → a «ZZ CARPETAS VACIAS PARA REVISAR/<cliente>/».
// Uso: node revision.mjs [--solo-verificar]

let token = null, vence = 0;
async function obtenerToken() {
  if (token && Date.now() < vence) return token;
  const r = await fetch(`https://login.microsoftonline.com/${process.env.AZURE_TENANT_ID}/oauth2/v2.0/token`, { method: 'POST', headers: { 'content-type': 'application/x-www-form-urlencoded' }, body: new URLSearchParams({ client_id: process.env.AZURE_CLIENT_ID, client_secret: process.env.AZURE_CLIENT_SECRET, scope: 'https://graph.microsoft.com/.default', grant_type: 'client_credentials' }) });
  if (!r.ok) throw new Error(`token: ${r.status}`);
  const j = await r.json(); token = j.access_token; vence = Date.now() + (j.expires_in - 300) * 1000; return token;
}
const carpetasOk = new Set(); const permitidos = new Set();
async function llamar(metodo, ruta, { cuerpo, ifMatch } = {}) {
  if (!['GET', 'POST', 'PATCH'].includes(metodo)) throw new Error(`BLOQUEADO: ${metodo}`);
  if (metodo !== 'GET') {
    if (!CORREGIR) throw new Error('BLOQUEADO: solo verificar');
    if (metodo === 'POST') { const m = /\/items\/([^/]+)\/children$/.exec(ruta); if (!m || !carpetasOk.has(m[1]) || !cuerpo.folder || cuerpo['@microsoft.graph.conflictBehavior'] !== 'fail') throw new Error('BLOQUEADO POST'); }
    if (metodo === 'PATCH') { const m = /\/items\/([^/?]+)\?@microsoft\.graph\.conflictBehavior=fail$/.exec(ruta); if (!m || !permitidos.has(m[1]) || !carpetasOk.has(cuerpo.parentReference?.id) || Object.keys(cuerpo).some((k) => !['name', 'parentReference'].includes(k))) throw new Error('BLOQUEADO PATCH'); }
  }
  for (let n = 1; n <= 8; n++) {
    try {
      const h = { authorization: `Bearer ${await obtenerToken()}` }; if (cuerpo) h['content-type'] = 'application/json'; if (ifMatch) h['if-match'] = ifMatch;
      const r = await fetch(ruta.startsWith('http') ? ruta : `https://graph.microsoft.com/v1.0${ruta}`, { method: metodo, headers: h, body: cuerpo ? JSON.stringify(cuerpo) : undefined, signal: AbortSignal.timeout(120_000) });
      if (r.status === 429 || r.status >= 500) { await new Promise((s) => setTimeout(s, Number(r.headers.get('retry-after') ?? 0) * 1000 || 3000 * n)); continue; }
      return r;
    } catch (e) { if (n === 8) throw e; await new Promise((s) => setTimeout(s, 3000 * n)); }
  }
}

async function inventario() {
  const items = new Map(); let url = `/drives/${driveId}/root/delta?$select=id,name,size,file,folder,parentReference,deleted,root,eTag`; let idRaiz = null;
  while (url) { const p = await jget(url); for (const it of p.value) { if (it.deleted) continue; if (it.root) idRaiz = it.id; items.set(it.id, { id: it.id, padre: it.parentReference?.id, nombre: it.name, carpeta: !!it.folder, hijos: it.folder?.childCount ?? 0, tam: it.size, hash: it.file?.hashes?.quickXorHash ?? null }); } url = p['@odata.nextLink'] ?? null; }
  const rutas = new Map();
  const ruta = (id) => { if (rutas.has(id)) return rutas.get(id); const it = items.get(id); const r = it.padre === idRaiz || !items.has(it.padre) ? it.nombre : `${ruta(it.padre)}/${it.nombre}`; rutas.set(id, r); return r; };
  for (const it of items.values()) it.ruta = ruta(it.id);
  return items;
}
const idDe = new Map();
async function carpeta(rel) {
  if (idDe.has(rel)) return idDe.get(rel);
  let r = await llamar('GET', `/drives/${driveId}/root:/${enc(rel)}?$select=id`); let id;
  if (r.ok) id = (await r.json()).id;
  else if (r.status === 404) {
    const padre = await carpeta(path.posix.dirname(rel));
    r = await llamar('POST', `/drives/${driveId}/items/${padre}/children`, { cuerpo: { name: path.posix.basename(rel), folder: {}, '@microsoft.graph.conflictBehavior': 'fail' } });
    if (r.status === 409) id = (await jget(`/drives/${driveId}/root:/${enc(rel)}?$select=id`)).id;
    else if (r.ok) { id = (await r.json()).id; fs.appendFileSync(DIARIO, JSON.stringify({ tipo: 'carpetaCreada', origen: 'revision', id, ruta: rel, ts: new Date().toISOString() }) + '\n'); }
    else throw new Error(`crear ${rel}: ${r.status}`);
  } else throw new Error(`GET ${rel}: ${r.status}`);
  carpetasOk.add(id); idDe.set(rel, id); return id;
}
async function mover(id, rutaDestino, motivo, cod) {
  const it = await jget(`/drives/${driveId}/items/${id}?$select=id,name,parentReference,eTag,folder`);
  const destinoId = await carpeta(path.posix.dirname(rutaDestino));
  permitidos.add(id);
  const nombre = path.posix.basename(rutaDestino);
  const p = await llamar('PATCH', `/drives/${driveId}/items/${id}?@microsoft.graph.conflictBehavior=fail`, { cuerpo: { name: nombre, parentReference: { id: destinoId } }, ifMatch: it.eTag });
  if (!p.ok) return `${p.status}`;
  fs.appendFileSync(DIARIO, JSON.stringify({ tipo: it.folder ? 'carpetaVaciaApartada' : 'movido', origen: 'revision', motivo, id, cod, deCarpetaId: it.parentReference.id, deNombre: it.name, aCarpetaId: destinoId, aNombre: nombre, aRuta: rutaDestino, ts: new Date().toISOString() }) + '\n');
  return null;
}

// ------------------------------------------------------------------ una ronda
function estructuraValida(rel, cod, esCarpeta) {
  // rel: ruta dentro de la carpeta del cliente. Devuelve null si cumple el 999, o el motivo.
  const s = rel.split('/');
  const n1 = new RegExp(`^${cod} (LEGAL Y SOCIETARIO|TRIBUTARIO|ZZ A CLASIFICAR|PERIODO (\\d{4}))$`).exec(s[0]);
  if (!n1) return 'en el primer nivel del cliente solo van las 4 carpetas del manual';
  if (s.length === 1) return esCarpeta ? null : 'archivo suelto en el primer nivel';
  const zona = n1[1];
  if (zona === 'ZZ A CLASIFICAR') return null; // cualquier cosa, a propósito
  if (zona === 'TRIBUTARIO') return s.length === 2 && !esCarpeta ? null : 'TRIBUTARIO no lleva subcarpetas';
  if (zona === 'LEGAL Y SOCIETARIO') return s.length === 2 || (/^ASAMBLEA \d{4}$/.test(s[1]) && (s.length === 3 && !esCarpeta || s.length === 2)) ? null : 'en LEGAL Y SOCIETARIO solo hay subcarpetas ASAMBLEA AAAA';
  const anio = n1[2];
  if (!['DOCUMENTOS CONTABLES', 'DOCUMENTOS LABORALES', 'DOCUMENTOS VARIOS'].includes(s[1])) return 'dentro de PERIODO solo van DOCUMENTOS CONTABLES, LABORALES y VARIOS';
  if (s.length === 2) return esCarpeta ? null : 'archivo suelto dentro de PERIODO';
  if (s[1] === 'DOCUMENTOS VARIOS') return null; // admite subcarpetas (FACTURACION EFFORT, pedidos y paquetes)
  if (s[1] === 'DOCUMENTOS LABORALES') return ['IPS', 'MTESS'].includes(s[2]) && (s.length === 3 ? esCarpeta : s.length === 4 && !esCarpeta) ? null : 'en DOCUMENTOS LABORALES solo IPS y MTESS';
  const mes = /^(0[1-9]|1[0-2]) ([A-Z]+)$/.exec(s[2]);
  const okMes = mes && MESES[+mes[1] - 1] === mes[2];
  if (!okMes && s[2] !== `CIERRE ${anio}`) return 'en DOCUMENTOS CONTABLES solo MM MES y CIERRE AAAA';
  if (s.length === 3) return esCarpeta ? null : 'archivo suelto en DOCUMENTOS CONTABLES';
  if (s.length === 4) return esCarpeta ? (okMes && (/^NC (EMITIDAS|RECIBIDAS) \d{6}$/.test(s[3]) || s[3] === 'IMPUTACION MASIVA') ? null : 'subcarpeta no prevista en el mes') : null;
  return s.length === 5 && !esCarpeta && okMes ? null : 'demasiados niveles';
}

async function ronda(numero) {
  const items = await inventario();
  const pr = { plan: [], copia: [], estructura: [], duplicados: [], vacias: [], modificadosPorEffort: [] };
  // 1. contra el PLAN
  let enLugar = 0;
  for (const f of plan) {
    if (EXCLUIR.has(f.id)) continue;
    const it = items.get(f.id);
    const esperado = `${BASE}/${f.destinoRel}`;
    if (!it) { pr.plan.push({ id: f.id, problema: 'ya no existe en OneDrive', esperado: f.destinoRel }); continue; }
    if (it.ruta === esperado) { enLugar++; continue; }
    // Modificado por EFFORT después del plan: no se fuerza a su destino (la estructura igual se controla abajo).
    if (f.hash && it.hash && it.hash !== f.hash) { pr.modificadosPorEffort.push({ ruta: it.ruta, delPlan: true }); continue; }
    if (f.accion === 'DUPLICADO') pr.duplicados.push({ id: f.id, ahora: it.ruta, esperado, cod: f.cod, hash: it.hash, hashPlan: f.hash });
    else pr.plan.push({ id: f.id, ahora: it.ruta, esperado, cod: f.cod, hash: it.hash, hashPlan: f.hash });
  }
  // 2. contra la COPIA DE SEGURIDAD (nada falta; el contenido es el mismo)
  for (const a of copia) {
    const it = items.get(a.id);
    if (!it) pr.copia.push({ id: a.id, ruta: a.ruta, problema: 'está en la copia de seguridad y no en OneDrive' });
    else if (a.hash && it.hash && it.hash !== a.hash) pr.modificadosPorEffort.push({ ruta: it.ruta });
  }
  // 3. ESTRUCTURA DEL 999 en cada cliente; carpetas vacías
  const clientes = [...items.values()].filter((it) => it.carpeta && it.ruta.startsWith(`${BASE}/`) && it.ruta.split('/').length === 3 && /^\d{3}(?!\d)/.test(it.nombre) && !it.nombre.startsWith('999'));
  for (const c of clientes) {
    const cod = c.nombre.slice(0, 3);
    for (const it of items.values()) {
      if (!it.ruta.startsWith(c.ruta + '/') || EXCLUIR.has(it.id)) continue;
      const rel = it.ruta.slice(c.ruta.length + 1);
      if (it.carpeta && it.hijos === 0 && rel !== `${cod} ZZ A CLASIFICAR`) { pr.vacias.push({ id: it.id, ruta: it.ruta, cliente: c.nombre, cod }); continue; }
      const f = planPorId.get(it.id);
      if (f && it.ruta === `${BASE}/${f.destinoRel}`) continue; // donde dice el plan aprobado: no se toca
      const mal = estructuraValida(rel, cod, it.carpeta);
      if (mal && !it.carpeta) pr.estructura.push({ id: it.id, ruta: it.ruta, cliente: c.nombre, cod, motivo: mal, rel });
      else if (mal && it.carpeta && !/^\d{3} ZZ A CLASIFICAR/.test(rel)) pr.estructura.push({ id: it.id, ruta: it.ruta, cliente: c.nombre, cod, motivo: mal, rel, carpeta: true });
    }
  }
  // Un archivo fuera de lugar según el plan también queda fuera de la estructura: se cuenta una sola vez.
  const enPlan = new Set([...pr.plan, ...pr.duplicados].map((x) => x.id));
  const archivosMal = pr.estructura.filter((e) => !e.carpeta && !enPlan.has(e.id));
  const total = pr.plan.length + pr.copia.length + archivosMal.length + pr.duplicados.length + pr.vacias.length;
  log(`RONDA ${numero}: en su lugar según el plan ${enLugar}/${plan.length} · fuera de lugar ${pr.plan.length} · duplicados aún dentro de clientes ${pr.duplicados.length} · faltan respecto de la copia ${pr.copia.length} · archivos fuera de la estructura del 999 ${archivosMal.length} · carpetas fuera del 999 ${pr.estructura.length - archivosMal.length} · carpetas vacías ${pr.vacias.length} · (modificados por EFFORT después: ${pr.modificadosPorEffort.length})`);
  fs.writeFileSync(path.join(AQUI, `revision-ronda-${numero}.json`), JSON.stringify(pr, null, 1));
  // 4. CORRECCIONES
  let corregidos = 0; const noSePudo = pr.copia.map((c) => ({ ...c, motivo: 'falta en OneDrive: restaurar desde la copia lo decide una persona' }));
  if (CORREGIR) {
    for (const p of [...pr.plan.filter((x) => x.ahora), ...pr.duplicados]) {
      if (p.hashPlan && p.hash !== p.hashPlan) { noSePudo.push({ ...p, motivo: 'su contenido cambió desde el plan: no se mueve solo' }); continue; }
      const e = await mover(p.id, p.esperado, 'no estaba donde dice el plan', p.cod); if (e) noSePudo.push({ ...p, motivo: e }); else corregidos++;
    }
    const yaMovidos = new Set([...pr.plan, ...pr.duplicados].map((x) => x.id));
    for (const e of archivosMal) {
      if (yaMovidos.has(e.id)) continue;
      const destino = `${BASE}/${e.cliente}/${e.cod} ZZ A CLASIFICAR/${e.rel}`;
      const err = await mover(e.id, destino, `fuera de la estructura del 999: ${e.motivo}`, e.cod); if (err) noSePudo.push({ ...e, motivo: err }); else corregidos++;
    }
    for (const v of pr.vacias) {
      const destino = `${BASE}/${ZZ_VAC}/${v.cliente}/${v.ruta.slice(`${BASE}/${v.cliente}/`.length).replaceAll('/', ' - ')}`;
      const err = await mover(v.id, destino, 'carpeta vacía dentro del cliente', v.cod); if (err) noSePudo.push({ ...v, motivo: err }); else corregidos++;
    }
  }
  if (noSePudo.length) fs.writeFileSync(path.join(AQUI, `revision-ronda-${numero}-no-corregidos.json`), JSON.stringify(noSePudo, null, 1));
  log(`  correcciones: ${corregidos} · no se pudieron corregir solas: ${noSePudo.length}`);
  return { total, corregidos, noSePudo: noSePudo.length, pr };
}

log(`=== REVISIÓN ${CORREGIR ? 'con correcciones' : 'solo verificación'} (mínimo 3 rondas)${EXCLUIR.size ? ` · EXCLUIDOS a pedido de Daniel: ${[...EXCLUIR].join(', ')}` : ''} ===`);
let ultima = null;
for (let r = 1; r <= 8; r++) {
  ultima = await ronda(r);
  if (r >= 3 && ultima.total === 0 && ultima.corregidos === 0) { log(`=== REVISIÓN LIMPIA: ronda ${r} sin ningún problema (después de ${r} rondas) ===`); process.exit(0); }
  if (r >= 3 && ultima.total === ultima.noSePudo && ultima.corregidos === 0) { log(`=== REVISIÓN TERMINADA CON ${ultima.noSePudo} CASOS QUE NECESITAN UNA PERSONA: ver revision-ronda-${r}-no-corregidos.json ===`); process.exit(2); }
  await new Promise((s) => setTimeout(s, 120_000));
}
log('=== REVISIÓN: 8 rondas sin quedar limpia. Ver revision-ronda-8.json ===');
