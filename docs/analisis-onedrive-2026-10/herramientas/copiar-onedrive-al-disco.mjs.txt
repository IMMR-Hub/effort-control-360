// ------------------------------------------------------------------ escritura: solo dentro de DESTINO
function dentroDelDestino(p) {
  const r = path.resolve(p);
  if (r !== DESTINO && !r.startsWith(DESTINO + path.sep)) throw new Error(`BLOQUEADO: escritura fuera del destino: ${r}`);

}

// ------------------------------------------------------------------ el disco puede desaparecer un instante (USB): esperar, no caerse
const ERRORES_DE_DISCO = /^(ENOENT|EIO|EBUSY|ENODEV|EPERM|UNKNOWN|EINVAL|ENXIO|ENOTCONN)$/;
async function esperarDisco() {
  const limite = Date.now() + 30 * 60_000;
  while (Date.now() < limite) {
    try { if (fs.existsSync(DESTINO) && fs.existsSync('D:/')) { fs.accessSync(DESTINO, fs.constants.W_OK); return true; } } catch { /* sigue esperando */ }
    await new Promise((s) => setTimeout(s, 5000));
  }
  throw new Error('El disco D: no volvió en 30 minutos');
}
async function anotar(archivo, objeto) {
  for (let i = 1; i <= 8; i++) {

}

// ------------------------------------------------------------------ quickXorHash (el que calcula OneDrive para Business)
class QuickXor {
  constructor() { this.acc = new Uint8Array(160); this.pos = 0; this.largo = 0; }
  update(buf) {
    let p = this.pos; const acc = this.acc;
    for (let i = 0; i < buf.length; i++) { acc[p] ^= buf[i]; if (++p === 160) p = 0; }
    this.pos = p; this.largo += buf.length;
  }
  digest() {
    const rgb = new Uint8Array(20);
    for (let r = 0; r < 160; r++) {
      const v = this.acc[r]; if (!v) continue;
      const desde = (11 * r) % 160;
      for (let k = 0; k < 8; k++) if ((v >> k) & 1) { const b = (desde + k) % 160; rgb[b >> 3] ^= 1 << (b & 7); }
    }
    let L = BigInt(this.largo);
    for (let i = 0; i < 8; i++) { rgb[12 + i] ^= Number(L & 0xffn); L >>= 8n; }
    return Buffer.from(rgb).toString('base64');
  }
}

// ------------------------------------------------------------------ Microsoft Graph: SOLO GET
let token = null, vence = 0;
async function obtenerToken() {
  if (token && Date.now() < vence) return token;
  const r = await fetch(`https://login.microsoftonline.com/${process.env.AZURE_TENANT_ID}/oauth2/v2.0/token`, {
    method: 'POST', headers: { 'content-type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({ client_id: process.env.AZURE_CLIENT_ID, client_secret: process.env.AZURE_CLIENT_SECRET, scope: 'https://graph.microsoft.com/.default', grant_type: 'client_credentials' }),
  });
  if (!r.ok) throw new Error(`token: ${r.status}`);
  const j = await r.json(); token = j.access_token; vence = Date.now() + (j.expires_in - 300) * 1000; return token;
}
async function llamar(metodo, ruta, opciones = {}) {
  if (metodo !== 'GET') throw new Error(`BLOQUEADO: solo se permite GET hacia OneDrive (pidió ${metodo})`);
  for (let intento = 1; intento <= 6; intento++) {
    try {
      const r = await fetch(ruta.startsWith('http') ? ruta : `https://graph.microsoft.com/v1.0${ruta}`, {
        method: 'GET', headers: { authorization: `Bearer ${await obtenerToken()}` }, signal: AbortSignal.timeout(opciones.timeout ?? 120_000),
      });
      if (r.status === 429 || r.status >= 500) { const espera = Number(r.headers.get('retry-after') ?? 0) * 1000 || 2500 * intento; await new Promise((s) => setTimeout(s, espera)); continue; }
      return r;
    } catch (e) { if (intento === 6) throw e; await new Promise((s) => setTimeout(s, 3000 * intento)); }
  }
  throw new Error('demasiados reintentos');
}
async function json(ruta) { const r = await llamar('GET', ruta); if (!r.ok) throw new Error(`GET ${ruta.slice(0, 90)} → ${r.status}`); return r.json(); }

// ------------------------------------------------------------------ descarga de un archivo, con verificación
async function bajar(a, destinoFinal) {
  const parte = dentroDelDestino(destinoFinal + '.parte');
  let esperasDeDisco = 0;
  for (let intento = 1; intento <= 4; intento++) {
    try {
      fs.mkdirSync(path.dirname(parte), { recursive: true });
      const r = await llamar('GET', `/drives/${driveId}/items/${a.id}/content`, { timeout: 30 * 60_000 });
      if (!r.ok) throw new Error(`HTTP ${r.status}`);
      const qx = new QuickXor(); let n = 0;
      const medir = new Transform({ transform(trozo, _e, cb) { qx.update(trozo); n += trozo.length; cb(null, trozo); } });
      if (r.body) await pipeline(Readable.fromWeb(r.body), medir, fs.createWriteStream(parte)); else fs.writeFileSync(parte, Buffer.alloc(0));
      const h = qx.digest();
      if (n !== a.tam) throw new Error(`peso distinto: bajó ${n}, OneDrive dice ${a.tam}`);
      if (a.hash && h !== a.hash) throw new Error('la huella no coincide con la de Microsoft');
      fs.renameSync(parte, dentroDelDestino(destinoFinal));
      if (a.mod) { const t = new Date(a.mod); fs.utimesSync(destinoFinal, t, t); }
      return { ok: true, hash: h, sinHashDeOrigen: !a.hash };
    } catch (e) {
      try { fs.rmSync(parte, { force: true }); } catch { /* nada */ }
      if (e && typeof e.code === 'string' && ERRORES_DE_DISCO.test(e.code) && esperasDeDisco < 20) {
        // Falló el DISCO, no el archivo ni la red: se espera a que vuelva y se reintenta sin gastar un intento.
        esperasDeDisco++; intento--; await esperarDisco(); continue;
      }
      if (intento === 4) return { ok: false, error: String(e.message ?? e).slice(0, 160) };
      await new Promise((s) => setTimeout(s, 2500 * intento));
    }
  }
}
