param([string]$lista)
# Dibuja las fotos de muestra (JPEG) con System.Drawing. Solo escribe en la carpeta local de preparación.
Add-Type -AssemblyName System.Drawing
$items = Get-Content -Raw -Encoding UTF8 $lista | ConvertFrom-Json
foreach ($it in $items) {
  $bmp = New-Object System.Drawing.Bitmap 900, 600
  $g = [System.Drawing.Graphics]::FromImage($bmp)
  $g.SmoothingMode = 'AntiAlias'
  $g.TextRenderingHint = 'AntiAlias'
  $g.Clear([System.Drawing.Color]::FromArgb(245, 240, 232))
  $pen = New-Object System.Drawing.Pen ([System.Drawing.Color]::FromArgb(107, 31, 78)), 6
  $g.DrawRectangle($pen, 12, 12, 876, 576)
  $titulo = New-Object System.Drawing.Font 'Calibri', 26, ([System.Drawing.FontStyle]::Bold)
  $normal = New-Object System.Drawing.Font 'Calibri', 20
  $marca = [System.Drawing.Brushes]::Black
  $g.DrawString('FOTO DE MUESTRA', $titulo, [System.Drawing.Brushes]::DarkRed, 40, 36)
  $y = 100
  $i = 0
  foreach ($linea in $it.lineas) {
    $fuente = $normal
    if ($i -eq 0) { $fuente = $titulo }
    $rect = New-Object System.Drawing.RectangleF 40, $y, 820, 140
    $g.DrawString([string]$linea, $fuente, $marca, $rect)
    $medida = $g.MeasureString([string]$linea, $fuente, 820)
    $y += [int]$medida.Height + 14
    $i++
  }
  $codec = [System.Drawing.Imaging.ImageCodecInfo]::GetImageEncoders() | Where-Object { $_.MimeType -eq 'image/jpeg' }
  $params = New-Object System.Drawing.Imaging.EncoderParameters 1
  $params.Param[0] = New-Object System.Drawing.Imaging.EncoderParameter ([System.Drawing.Imaging.Encoder]::Quality), 80L
  New-Item -ItemType Directory -Force -Path (Split-Path $it.ruta) | Out-Null
  $bmp.Save($it.ruta, $codec, $params)
  $g.Dispose(); $bmp.Dispose()
}
Write-Output ("fotos: " + $items.Count)
