// --- RUC con dígito verificador (módulo 11 de la DNIT; verificado con 3 RUC reales del piloto)
function dv(rucSinDv) {
  let total = 0, k = 2;
  for (const d of [...rucSinDv].reverse()) { total += Number(d) * k; k = k >= 11 ? 2 : k + 1; }
  const resto = total % 11;
  return resto > 1 ? 11 - resto : 0;
}
for (const [r, esperado] of [['80003112', 1], ['80082006', 1], ['80119631', 0]]) if (dv(r) !== esperado) throw new Error('DV mal calculado');

    const m = rucs.get(c.codigo);
    if (m) {
      const orden = [...m].sort((a, b) => b[1] - a[1]); const total = [...m.values()].reduce((s, x) => s + x, 0);
      const [top, n] = orden[0];
      ruc = `${top}-${dv(top)}`;
      notaRuc = n / total >= 0.8 ? `Sale de ${n} de ${total} declaraciones de su carpeta.` : `DUDOSO: ${n} de ${total} declaraciones dicen este RUC; hay otros (${orden.slice(1, 3).map((x) => x[0]).join(', ')}).`;
    } else notaRuc = 'No pude deducirlo: falta cargarlo.';
  }
  const n = ruc ? Number(ruc.split('-')[0]) : null;
  const tipo = p ? 'Jurídica' : n !== null ? (n >= 80000000 ? 'Jurídica' : 'Física') : JURIDICA.test(nombre) ? 'Jurídica' : 'Física';

  const hallados = f ? [...f].sort((a, b) => b[1] - a[1]).map(([k, v]) => `Form. ${k}${FORMULARIOS[k] ? ` (${FORMULARIOS[k]})` : ''} ×${v}`).join(', ') : '';

  return { codigo: c.codigo, nombre, archivos: c.archivos, ultimo: c.ultimo, ruc, notaRuc, tipo, hallados, actividad, piloto: !!p };
});
