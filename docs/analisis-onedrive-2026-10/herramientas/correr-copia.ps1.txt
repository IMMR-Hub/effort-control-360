[Energia.Windows]::SetThreadExecutionState(0x80000001) | Out-Null   # ES_CONTINUOUS | ES_SYSTEM_REQUIRED
$aqui = 'C:\Users\Daniel\AppData\Local\Temp\claude\C--Users-Daniel-NexusFlow-AI\cdba12b1-63af-4d45-a032-bade731625cb\scratchpad\copia'
$log = "$aqui\corrida2.log"
function Anotar($t) { Add-Content -Path $log -Value $t -Encoding utf8 }
Anotar "INICIO $(Get-Date -Format s)"
& node "$aqui\copiar.mjs" --verificar 2>&1 | ForEach-Object { Anotar ([string]$_) }
