// ---------------------------------------------------------------- mensual (documentos base de cada mes)
function mes(anio, n) {
  const dir = `PERIODO ${anio}/DOCUMENTOS CONTABLES/${mm(n)} ${MES[n - 1]}`;
  const p = `${mm(n)}${anio}`;
  const m = `${MES[n - 1].toLowerCase()} de ${anio}`;
  add(dir, `DDJJ IVA ${p} ${C}.pdf`, `Declaración jurada de IVA presentada, formulario 120 (${m})`);
  add(dir, `TALON RG 90 ${p} ${C}.pdf`, `Comprobante de presentación de la RG 90, formulario 241 (${m})`);
  add(dir, `RG COMPRAS ${p} ${C}.xlsx`, `Planilla RG 90 de compras (${m})`);
  add(dir, `RG VENTAS ${p} ${C}.xlsx`, `Planilla RG 90 de ventas (${m})`);
  add(dir, `CALCULO IVA ${p} ${C}.xlsx`, `Cálculo del IVA (${m})`);
  add(dir, `LIBRO COMPRAS ${p} ${C}.pdf`, `Libro de compras del SIGA (${m})`);
  add(dir, `LIBRO VENTAS ${p} ${C}.pdf`, `Libro de ventas del SIGA (${m})`);
  add(dir, `BOLETA DE PAGO IVA ${p} ${C}.pdf`, `Boleta de pago del IVA (${m})`);
  add(dir, `EXTRACTO ITAU ${p} ${C}.pdf`, `Extracto bancario del Itaú (${m})`);
  return { dir, p };
}
for (let n = 1; n <= 12; n++) mes(2026, n);

// ---------------------------------------------------------------- cierre anual
function cierre(anio) {
  const dir = `PERIODO ${anio}/DOCUMENTOS CONTABLES/CIERRE ${anio}`;
  add(dir, `DDJJ IRE ${anio} ${C}.pdf`, `Declaración jurada del IRE presentada, formulario 500 (ejercicio ${anio})`);
  add(dir, `EEFF ${anio} ${C}.pdf`, `Estados financieros firmados (ejercicio ${anio})`);
  add(dir, `EEFF ${anio} PRESENTADO FORM 158 ${C}.pdf`, `Constancia de presentación de los estados financieros ante la DNIT, formulario 158 (ejercicio ${anio})`);
  add(dir, `CALCULO IRE ${anio} ${C}.xlsx`, `Cálculo del IRE del cierre (ejercicio ${anio})`);
  add(dir, `BOLETA DE PAGO IRE ${anio} ${C}.pdf`, `Boleta de pago del IRE (ejercicio ${anio})`);
}
cierre(2026);

// ---------------------------------------------------------------- el principio del período: diciembre de 2025 y su cierre viven en PERIODO 2025
mes(2025, 12);
cierre(2025);
