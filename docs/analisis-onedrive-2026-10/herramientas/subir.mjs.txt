// Protecciones por construcción (no dependen de que alguien tenga cuidado):
//   · Solo se usan GET, PUT y POST. No existe ninguna rama de PATCH, DELETE, copy ni move.
//   · Todo PUT va a una ruta dentro de BASE; todo POST crea una carpeta nueva dentro de BASE
//     (o la carpeta raíz del cliente, directamente dentro de CLIENTES).
//   · conflictBehavior=fail en todo: nunca se pisa nada que ya exista.
//   · Se aborta si ya existe cualquier cosa que empiece con «999» en CLIENTES.
//   · Sin --aplicar solo simula: no hace ninguna escritura.
// Los secretos se leen del .env al proceso y van en el cuerpo de un fetch: no se imprimen.
import { readFileSync, writeFileSync } from 'node:fs';
import { createRequire } from 'node:module';
import path from 'node:path';

const APLICAR = process.argv.includes('--aplicar');
const AQUI = path.dirname(new URL(import.meta.url).pathname.replace(/^\/([A-Za-z]:)/, '$1'));
const RAIZ_REPO = 'C:/Users/Daniel/NexusFlow AI/effort-control-360';
const plan = createRequire(import.meta.url)('./plan.cjs');

for (const linea of readFileSync(`${RAIZ_REPO}/.env`, 'utf8').split('\n')) {
  const m = linea.match(/^([A-Z_][A-Z0-9_]*)=(.*)$/);
  if (m && !process.env[m[1]]) process.env[m[1]] = m[2].trim().replace(/^["']|["']$/g, '');
}

const PADRE = 'CLIENTES EFFORT E.A.S/CLIENTES';
const BASE = `${PADRE}/${plan.RAIZ}`;
const METODOS = new Set(['GET', 'PUT', 'POST']);
const enc = (ruta) => ruta.split('/').map(encodeURIComponent).join('/');

let token = null, vence = 0;
async function obtenerToken() {
  if (token && Date.now() < vence) return token;
  const r = await fetch(`https://login.microsoftonline.com/${process.env.AZURE_TENANT_ID}/oauth2/v2.0/token`, {
    method: 'POST', headers: { 'content-type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({ client_id: process.env.AZURE_CLIENT_ID, client_secret: process.env.AZURE_CLIENT_SECRET, scope: 'https://graph.microsoft.com/.default', grant_type: 'client_credentials' }),
  });
  if (!r.ok) throw new Error(`token: ${r.status}`);
  const j = await r.json(); token = j.access_token; vence = Date.now() + (j.expires_in - 300) * 1000; return token;
}

/** ÚNICA puerta de salida hacia Graph. Rechaza todo lo que no sea lo autorizado. */
async function llamar(metodo, ruta, { cuerpo, tipo, escritura } = {}) {
  if (!METODOS.has(metodo)) throw new Error(`Método no permitido: ${metodo}`);
  if (metodo !== 'GET') {
    const decodificada = decodeURIComponent(ruta);
    const dentroDeBase = decodificada.includes(`root:/${BASE}/`) || decodificada.includes(`root:/${BASE}:`);
    const esCrearRaiz = metodo === 'POST' && decodificada.endsWith(`root:/${PADRE}:/children`) && cuerpo?.name === plan.RAIZ && cuerpo?.folder;
    if (!dentroDeBase && !esCrearRaiz) throw new Error(`BLOQUEADO: escritura fuera de «${plan.RAIZ}»: ${decodificada}`);
    if (!APLICAR) throw new Error('BLOQUEADO: simulación, no se escribe');
    if (metodo === 'PUT' && !/[?&]@microsoft\.graph\.conflictBehavior=fail/.test(ruta)) throw new Error('BLOQUEADO: PUT sin conflictBehavior=fail');
    if (metodo === 'POST' && cuerpo['@microsoft.graph.conflictBehavior'] !== 'fail') throw new Error('BLOQUEADO: POST sin conflictBehavior=fail');
  }
  for (let intento = 1; intento <= 5; intento++) {
    try {
      const r = await fetch(ruta.startsWith('http') ? ruta : `https://graph.microsoft.com/v1.0${ruta}`, {
        method: metodo,
        headers: { authorization: `Bearer ${await obtenerToken()}`, ...(tipo ? { 'content-type': tipo } : {}) },
        body: cuerpo === undefined ? undefined : (Buffer.isBuffer(cuerpo) ? cuerpo : JSON.stringify(cuerpo)),
      });
      if (r.status === 429 || r.status >= 500) { await new Promise((s) => setTimeout(s, 2000 * intento)); continue; }
      return r;
    } catch (e) { if (intento === 5) throw e; await new Promise((s) => setTimeout(s, 2500 * intento)); }
  }
  throw new Error('demasiados reintentos');
}

const usuario = process.env.ONEDRIVE_USUARIO_ORIGEN ?? 'lsosa@effort.com.py';
const driveId = process.env.AZURE_DRIVE_ID_ORIGEN ?? (await (await llamar('GET', `/users/${encodeURIComponent(usuario)}/drive?$select=id`)).json()).id;

// ---- 1) Comprobaciones previas (solo lectura)
async function hijos(rutaOId, porId) {
  const base = porId ? `/drives/${driveId}/items/${rutaOId}/children` : `/drives/${driveId}/root:/${enc(rutaOId)}:/children`;
  let url = `${base}?$select=id,name,size,folder,file&$top=200`; const out = [];
  while (url) {
    const r = await llamar('GET', url); if (!r.ok) throw new Error(`GET ${r.status} ${url.slice(0, 90)}`);
    const j = await r.json(); out.push(...j.value); url = j['@odata.nextLink'] ?? null;
  }
  return out;
}
const antes = await hijos(PADRE, false);
const con999 = antes.filter((x) => x.name.startsWith('999'));
console.log(`CLIENTES tiene ${antes.length} elementos; con «999»: ${con999.length}`);
if (con999.length) throw new Error(`ABORTADO: ya existe algo que empieza con 999: ${con999.map((x) => x.name).join(', ')}`);
const nombresAntes = antes.map((x) => x.name).sort();

// ---- 2) Lo que se va a crear
const carpetas = new Set();
for (const a of plan.archivos) { const partes = a.dir.split('/'); for (let i = 1; i <= partes.length; i++) carpetas.add(partes.slice(0, i).join('/')); }
for (const v of plan.carpetasVacias) carpetas.add(v);
const listaCarpetas = [...carpetas].sort((x, y) => x.split('/').length - y.split('/').length || x.localeCompare(y));
console.log(`Se crearía: 1 carpeta de cliente («${plan.RAIZ}»), ${listaCarpetas.length} subcarpetas y ${plan.archivos.length} archivos.`);
if (!APLICAR) {
  console.log('SIMULACIÓN: no se escribió nada. Subcarpetas:');
  console.log(listaCarpetas.map((c) => '  ' + c).join('\n'));
  process.exit(0);
}

// ---- 3) Creación
const registro = { inicio: new Date().toISOString(), carpetas: [], archivos: [], errores: [] };
const idDeCarpeta = new Map();
{
  const r = await llamar('POST', `/drives/${driveId}/root:/${enc(PADRE)}:/children`, { tipo: 'application/json', cuerpo: { name: plan.RAIZ, folder: {}, '@microsoft.graph.conflictBehavior': 'fail' } });
  if (r.status !== 201) throw new Error(`No se pudo crear la carpeta del cliente: ${r.status} ${await r.text()}`);
  const j = await r.json(); idDeCarpeta.set('', j.id); registro.carpetas.push(plan.RAIZ);
  console.log(`Creada «${plan.RAIZ}»`);
}
for (const c of listaCarpetas) {
  const partes = c.split('/'); const padre = partes.slice(0, -1).join('/'); const nombre = partes.at(-1);
  // Por RUTA (no por id): así la puerta de salida puede comprobar que el padre está dentro de «999 CLIENTE MUESTRA».
  const rutaPadre = padre ? `${BASE}/${padre}` : BASE;
  const r = await llamar('POST', `/drives/${driveId}/root:/${enc(rutaPadre)}:/children`, { tipo: 'application/json', cuerpo: { name: nombre, folder: {}, '@microsoft.graph.conflictBehavior': 'fail' } });
  if (r.status !== 201) throw new Error(`No se pudo crear «${c}»: ${r.status} ${await r.text()}`);
  registro.carpetas.push(`${plan.RAIZ}/${c}`);
}
console.log(`Creadas ${listaCarpetas.length} subcarpetas`);

let hechos = 0, i = 0;
async function trabajador() {
  while (i < plan.archivos.length) {
    const a = plan.archivos[i++];
    const rutaOneDrive = `${BASE}/${a.dir}/${a.nombre}`;
    const cuerpo = readFileSync(path.join(AQUI, 'archivos', a.dir, a.nombre));
    const url = `/drives/${driveId}/root:/${enc(rutaOneDrive)}:/content?@microsoft.graph.conflictBehavior=fail`;
    let listo = false;
    for (let intento = 1; intento <= 3 && !listo; intento++) {
      const r = await llamar('PUT', url, { tipo: 'application/octet-stream', cuerpo });
      if (r.status === 201 || r.status === 200) { registro.archivos.push({ ruta: rutaOneDrive, bytes: cuerpo.length }); listo = true; }
      else if (r.status === 409) {
        // ¿Quedó creado en un intento anterior que se cortó? Se acepta solo si es exactamente el mismo peso.
        const g = await llamar('GET', `/drives/${driveId}/root:/${enc(rutaOneDrive)}?$select=size`);
        if (g.ok && (await g.json()).size === cuerpo.length) { registro.archivos.push({ ruta: rutaOneDrive, bytes: cuerpo.length, nota: 'ya estaba' }); listo = true; } else break;
      } else await new Promise((s) => setTimeout(s, 1500 * intento));
    }
    if (!listo) registro.errores.push(rutaOneDrive);
    if (++hechos % 40 === 0) console.log(`  ${hechos}/${plan.archivos.length}`);
  }
}
await Promise.all(Array.from({ length: 4 }, trabajador));
registro.fin = new Date().toISOString();
writeFileSync(path.join(AQUI, 'registro.json'), JSON.stringify(registro, null, 1));
console.log(`Subidos ${registro.archivos.length}/${plan.archivos.length} archivos; errores: ${registro.errores.length}`);

// ---- 4) Verificación: lo que hay en OneDrive == lo previsto, y nada más cambió en CLIENTES
const previsto = new Map(plan.archivos.map((a) => [`${a.dir}/${a.nombre}`, readFileSync(path.join(AQUI, 'archivos', a.dir, a.nombre)).length]));
const encontrado = new Map(); const carpetasEnc = new Set();
async function recorrer(id, rel) {
  for (const x of await hijos(id, true)) {
    const r = rel ? `${rel}/${x.name}` : x.name;
    if (x.folder) { carpetasEnc.add(r); await recorrer(x.id, r); } else encontrado.set(r, x.size);
  }
}
await recorrer(idDeCarpeta.get(''), '');
const faltan = [...previsto.keys()].filter((k) => !encontrado.has(k));
const sobran = [...encontrado.keys()].filter((k) => !previsto.has(k));
const pesoMal = [...previsto].filter(([k, v]) => encontrado.has(k) && encontrado.get(k) !== v).map(([k]) => k);
const carpetasFaltan = [...carpetas].filter((c) => !carpetasEnc.has(c));
const despues = (await hijos(PADRE, false)).map((x) => x.name).sort();
const nuevos = despues.filter((n) => !nombresAntes.includes(n)); const perdidos = nombresAntes.filter((n) => !despues.includes(n));
console.log(JSON.stringify({ archivosEnOneDrive: encontrado.size, carpetasEnOneDrive: carpetasEnc.size, faltan: faltan.length, sobran: sobran.length, pesoDistinto: pesoMal.length, carpetasFaltan: carpetasFaltan.length, clientesAntes: nombresAntes.length, clientesDespues: despues.length, nuevosEnCLIENTES: nuevos, perdidosEnCLIENTES: perdidos }));
