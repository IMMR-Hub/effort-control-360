function consultar(sql) {
  for (let i = 1; i <= 5; i++) {
    try {
      const txt = execFileSync('node', ['scripts/consultar-produccion.mjs', sql], { cwd: RAIZ, encoding: 'utf8', maxBuffer: 64 * 1024 * 1024 });
      return JSON.parse(txt.replace(/"(-?\d+)n"/g, '$1'));
    } catch (e) {
      if (i === 5) throw e;
    }
  }
}

const venc = consultar(`SELECT v.id, v.cliente_id, c.nombre AS cliente, v.tipo_documento::text AS tipo, v.descripcion, to_char(v.fecha_vencimiento,'YYYY-MM-DD') AS fv, v.periodo
  FROM vencimiento v JOIN cliente c ON c.id = v.cliente_id
  WHERE v.estado::text NOT IN ('PRESENTADO','NO_APLICA') AND v.periodo IS NOT NULL`);
const liq = consultar(`SELECT cliente_id, periodo, comprobantes_compras, comprobantes_ventas FROM liquidacion_iva_rg90`);
